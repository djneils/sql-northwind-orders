NORTHWIND ORDERS DATABASE - SOLUTIONS
======================================

ACTIVITY 1 - Basic SELECT
--------------------------
Q1: SELECT * FROM Customers WHERE Country = 'UK';
Q2: SELECT * FROM Customers WHERE Country = 'UK' ORDER BY City;
Q3: SELECT FirstName, LastName, BirthDate FROM Employees WHERE LastName = 'Callahan';
Q4: SELECT * FROM Suppliers ORDER BY Country;
Q5: SELECT * FROM Products WHERE Price < 20;
Q6: SELECT * FROM Products ORDER BY Price DESC;

ACTIVITY 2 - AND, OR, BETWEEN
-------------------------------
Q7: SELECT * FROM Customers WHERE Country = 'UK' OR Country = 'Germany';
Q8: SELECT * FROM Customers WHERE Country = 'UK' AND City != 'London';
Q9: SELECT * FROM Suppliers WHERE Country IN ('USA', 'Canada') ORDER BY SupplierName;
Q10: SELECT * FROM Products WHERE Price BETWEEN 20 AND 40;

ACTIVITY 3 - JOINs
--------------------
Q11: SELECT Products.* FROM Products JOIN Suppliers ON Products.SupplierID = Suppliers.SupplierID WHERE Suppliers.SupplierName = 'Leka Trading';
Q12: SELECT Orders.* FROM Orders JOIN Customers ON Orders.CustomerID = Customers.CustomerID WHERE Customers.CustomerName = 'Island Trading';
Q13: SELECT Products.* FROM Products JOIN Suppliers ON Products.SupplierID = Suppliers.SupplierID WHERE Suppliers.Country = 'UK';
Q14: SELECT Products.ProductName, Suppliers.SupplierName, Suppliers.Country FROM Products JOIN Suppliers ON Products.SupplierID = Suppliers.SupplierID WHERE Suppliers.Country IN ('USA', 'Canada');
Q15: SELECT DISTINCT Products.ProductName FROM Products JOIN OrderDetails ON Products.ProductID = OrderDetails.ProductID JOIN Orders ON OrderDetails.OrderID = Orders.OrderID JOIN Customers ON Orders.CustomerID = Customers.CustomerID WHERE Customers.CustomerName = 'Island Trading';
Q16: SELECT DISTINCT Shippers.ShipperName FROM Shippers JOIN Orders ON Shippers.ShipperID = Orders.ShipperID JOIN Customers ON Orders.CustomerID = Customers.CustomerID WHERE Customers.Country = 'Argentina';
Q17: SELECT DISTINCT Customers.CustomerName, Customers.ContactName FROM Customers JOIN Orders ON Customers.CustomerID = Orders.CustomerID JOIN Employees ON Orders.EmployeeID = Employees.EmployeeID WHERE Employees.FirstName = 'Robert' AND Employees.LastName = 'King' ORDER BY Customers.ContactName;
